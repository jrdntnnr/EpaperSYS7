cmake_minimum_required(VERSION 3.16)
project(papers3_presenter C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(paper_presenter components/paper_presenter/paper_presenter.c)
target_include_directories(paper_presenter PUBLIC components/paper_presenter/include)
target_compile_options(paper_presenter PRIVATE -Wall -Wextra -Werror)
add_executable(test_presenter tests/test_presenter.c)
target_link_libraries(test_presenter PRIVATE paper_presenter)
target_compile_options(test_presenter PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME presenter COMMAND test_presenter)

add_library(paper_surface components/paper_epd/paper_surface.c)
target_include_directories(paper_surface PUBLIC components/paper_epd/include)
target_link_libraries(paper_surface PUBLIC paper_presenter)
target_compile_options(paper_surface PRIVATE -Wall -Wextra -Werror)
add_executable(test_surface tests/test_surface.c)
target_link_libraries(test_surface PRIVATE paper_surface)
target_compile_options(test_surface PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME surface COMMAND test_surface)
set_tests_properties(presenter surface PROPERTIES TIMEOUT 30)
